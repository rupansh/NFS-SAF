cmake_minimum_required(VERSION 3.22)
project(nfssaf LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_TLS OFF CACHE STRING "" FORCE)
set(ENABLE_NFS4_2 ON CACHE BOOL "" FORCE)
set(ENABLE_MULTITHREADING OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/libnfs libnfs EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_library(nfs_storage STATIC storage.cpp renew_v4.c)
target_include_directories(nfs_storage PRIVATE ../vendor/libnfs/include ../vendor/libnfs/include/nfsc ${CMAKE_CURRENT_BINARY_DIR}/libnfs)
target_compile_definitions(nfs_storage PRIVATE HAVE_NFS4_2)
target_include_directories(nfs_storage PUBLIC . ../vendor/libnfs/include)
target_link_libraries(nfs_storage PUBLIC nfs Threads::Threads)
target_compile_options(nfs_storage PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
  add_library(nfssaf SHARED jni.cpp)
  target_link_libraries(nfssaf PRIVATE nfs_storage)
  target_link_options(nfssaf PRIVATE -Wl,-z,max-page-size=16384)
else()
  enable_testing()
  add_executable(storage_tests tests.cpp)
  target_link_libraries(storage_tests PRIVATE nfs_storage)
  add_test(NAME storage_unit COMMAND storage_tests)
endif()
